def display_coef(model, features):
    idx = np.argsort(np.abs(model.coef_))[::-1]

    for k, (f, c) in enumerate(zip(features[idx], model.coef_[idx]), start = 1):
        print(f'{k:>3}.{f:<40}{c:+5.4f}')
